# Create ERA5-based cyclone detection function
def create_cyclone_detection_task():
    """
    Create GEE export task for ERA5-based cyclone identification
    """
    
    # ERA5 pressure level data for cyclone detection
    era5_pressure = ee.ImageCollection('ECMWF/ERA5_LAND/HOURLY') \
                     .select(['surface_pressure', 'u_component_of_wind_10m', 'v_component_of_wind_10m']) \
                     .filterDate(config['historical_period']['start'], config['historical_period']['end']) \
                     .filterBounds(SIO_REGION)
    
    # CHIRPS precipitation for storm signature
    precipitation = ee.ImageCollection('UCSB-CHG/CHIRPS/DAILY') \
                     .filterDate(config['historical_period']['start'], config['historical_period']['end']) \
                     .filterBounds(SIO_REGION)
    
    # Sea surface temperature
    sst = ee.ImageCollection('NOAA/CDR/OISST/V2_1') \
           .select('sst') \
           .filterDate(config['historical_period']['start'], config['historical_period']['end']) \
           .filterBounds(SIO_REGION)
    
    print("📊 GEE Collections loaded:")
    print(f"   📉 ERA5 pressure/wind: {era5_pressure.size().getInfo()} images")
    print(f"   🌧️ CHIRPS precipitation: {precipitation.size().getInfo()} images")
    print(f"   🌊 OISST sea temperature: {sst.size().getInfo()} images")
    
    return era5_pressure, precipitation, sst

def find_repo_root(start: Path) -> Path:
    start = start.resolve()
    for p in [start] + list(start.parents):
        if (p / "requirements.txt").exists() and (p / "data").exists():
            return p
    for p in [start] + list(start.parents):
        if (p / "data").exists():
            return p
    raise RuntimeError("Could not locate repo root with data/")

def normalize(f):
    out = ee.Feature(f.geometry())

    if id_field:
        out = out.set("event_id", f.get(id_field))
    else:
        out = out.set("event_id", None)

    if country_field:
        out = out.set("country", f.get(country_field))
    else:
        out = out.set("country", None)

    if type_field:
        out = out.set("hazard_type", f.get(type_field))
    else:
        out = out.set("hazard_type", None)

    if subtype_field:
        out = out.set("hazard_subtype", f.get(subtype_field))
    else:
        out = out.set("hazard_subtype", None)

    if date_field:
        out = out.set("event_date_raw", f.get(date_field))
    else:
        out = out.set("event_date_raw", None)

    # Always include a centroid for easy H3 join later
    centroid = f.geometry().centroid(1)
    out = out.set({
        "longitude": centroid.coordinates().get(0),
        "latitude": centroid.coordinates().get(1),
    })

    return out

for cyclone, cols in CYCLONE_COLS.items():

impacts = {}

        # Convert to numeric

            temp[c] = pd.to_numeric(temp[c], errors='coerce').fillna(0)
        
        # Add cyclone name
        temp['cyclone'] = cyclone
        
        # Filter to wards with actual damage (Option B - strict labeling)

# Cell 3: Extract Extreme Rainfall Events (100k Sample - Ultra-Fast)
"""
Ultra-fast event extraction using vectorized operations and memory optimization.
Maintains all statistics while maximizing processing speed.
"""

import pandas as pd
import numpy as np
import rasterio
from pathlib import Path
import time
import gc
from datetime import datetime, timedelta

print('⚡ ULTRA-FAST EXTREME EVENTS EXTRACTION')
print('=' * 39)

# Self-contained configuration
BASE_DIR = Path('C:/Projects/ramangwana/risks/data/extreme_rainfall')
CHIRPS_DIR = BASE_DIR / '01_climate_baseline'
EVENTS_OUTPUT_DIR = BASE_DIR / 'preprocessed' / 'events'

# Sample configuration
SAMPLE_SIZE = 100000
PERCENTILE_THRESHOLD = 95
MIN_RAINFALL_THRESHOLD = 10
RANDOM_STATE = 42
EVENT_START_YEAR = 2000
EVENT_END_YEAR = 2024

# Load data
chirps_files = sorted(CHIRPS_DIR.glob('zimbabwe_chirps_daily_*.tif'))
relevant_chirps = [f for f in chirps_files if EVENT_START_YEAR <= int(f.stem.split('_')[-1]) <= EVENT_END_YEAR]
valid_pixels_df = pd.read_parquet(EVENTS_OUTPUT_DIR / 'all_valid_pixels.parquet')

print(f'⚡ Ultra-Fast Configuration:')
print(f'   Sample size: {SAMPLE_SIZE:,} pixels')
print(f'   CHIRPS years: {len(relevant_chirps)} files')
print(f'   Available pixels: {len(valid_pixels_df):,}')

# Output files
sample_events_file = EVENTS_OUTPUT_DIR / 'sample_extreme_events.parquet'

if sample_events_file.exists():
    print('✅ Sample events database already exists')
    sample_events = pd.read_parquet(sample_events_file)
    print(f'   📊 Events in database: {len(sample_events):,}')
else:
    start_time = time.time()
    
    # Sample pixels
    sample_pixels = valid_pixels_df.sample(n=SAMPLE_SIZE, random_state=RANDOM_STATE)
    
    # Convert to numpy arrays for vectorized operations
    pixel_rows = sample_pixels['chirps_row'].values
    pixel_cols = sample_pixels['chirps_col'].values
    pixel_h3_ids = sample_pixels['h3_id'].values
    pixel_lats = sample_pixels['center_lat'].values
    pixel_lngs = sample_pixels['center_lng'].values
    
    print(f'⚡ Processing {len(sample_pixels):,} pixels with vectorized operations...')
    
    # Ultra-fast data collection
    all_pixel_data = np.zeros((len(sample_pixels), len(relevant_chirps) * 365), dtype=np.float32)
    day_count = 0
    date_index = []
    
    for file_idx, chirps_file in enumerate(relevant_chirps):
        print(f'   ⚡ {chirps_file.name} ({file_idx+1}/{len(relevant_chirps)})...', end='')
        file_start = time.time()
        
        year = int(chirps_file.stem.split('_')[-1])
        
        with rasterio.open(chirps_file) as src:
            for band_idx in range(1, src.count + 1):
                # Read entire band once
                band_data = src.read(band_idx)
                
                # Vectorized pixel extraction (all pixels at once)
                day_values = band_data[pixel_rows, pixel_cols]
                all_pixel_data[:, day_count] = np.maximum(day_values, 0)
                
                # Store date
                try:
                    event_date = datetime(year, 1, 1) + timedelta(days=band_idx - 1)
                    date_index.append(event_date)
                except:
                    date_index.append(None)
                
                day_count += 1
                del band_data
        
        file_time = time.time() - file_start
        print(f' ({file_time:.1f}s)')
    
    # Trim to actual data
    all_pixel_data = all_pixel_data[:, :day_count]
    
    print(f'   ⚡ Data shape: {all_pixel_data.shape[0]:,} pixels × {all_pixel_data.shape[1]:,} days')
    
    # Ultra-fast threshold calculation (vectorized)
    print('⚡ Calculating thresholds (vectorized)...')
    
    # Filter out zero values for threshold calculation
    valid_data_mask = all_pixel_data > 0
    thresholds_95 = np.zeros(len(sample_pixels))
    thresholds_99 = np.zeros(len(sample_pixels))
    
    for i in range(len(sample_pixels)):
        pixel_data = all_pixel_data[i, valid_data_mask[i]]
        if len(pixel_data) >= 100:
            thresholds_95[i] = np.percentile(pixel_data, PERCENTILE_THRESHOLD)
            thresholds_99[i] = np.percentile(pixel_data, 99)
    
    # Ultra-fast event detection (vectorized)
    print('⚡ Detecting events (vectorized)...')
    
    events_list = []
    
    # Create threshold matrix for broadcasting
    threshold_95_matrix = np.tile(thresholds_95.reshape(-1, 1), (1, day_count))
    threshold_99_matrix = np.tile(thresholds_99.reshape(-1, 1), (1, day_count))
    
    # Vectorized event detection
    extreme_mask = (all_pixel_data >= threshold_95_matrix) & (all_pixel_data >= MIN_RAINFALL_THRESHOLD)
    severe_mask = extreme_mask & (all_pixel_data < threshold_99_matrix)
    ultra_extreme_mask = extreme_mask & (all_pixel_data >= threshold_99_matrix)
    
    # Extract events
    extreme_pixels, extreme_days = np.where(extreme_mask)
    
    print(f'⚡ Creating event records for {len(extreme_pixels):,} events...')
    
    for i in range(len(extreme_pixels)):
        pixel_idx = extreme_pixels[i]
        day_idx = extreme_days[i]
        
        if date_index[day_idx] is not None:
            rainfall = all_pixel_data[pixel_idx, day_idx]
            is_ultra_extreme = ultra_extreme_mask[pixel_idx, day_idx]
            
            events_list.append({
                'event_id': i,
                'h3_id': pixel_h3_ids[pixel_idx],
                'date': date_index[day_idx],
                'year': date_index[day_idx].year,
                'month': date_index[day_idx].month,
                'day_of_year': date_index[day_idx].timetuple().tm_yday,
                'rainfall_mm': float(rainfall),
                'magnitude': 'extreme' if is_ultra_extreme else 'severe',
                'magnitude_code': 2 if is_ultra_extreme else 1,
                'threshold_95th': thresholds_95[pixel_idx],
                'threshold_99th': thresholds_99[pixel_idx],
                'center_lat': pixel_lats[pixel_idx],
                'center_lng': pixel_lngs[pixel_idx]
            })
    
    # Create DataFrame
    sample_events = pd.DataFrame(events_list)
    sample_events.to_parquet(sample_events_file)
    
    total_time = time.time() - start_time
    
    print(f'\n⚡ Ultra-fast extraction complete!')
    print(f'   📊 Total events: {len(sample_events):,}')
    print(f'   📍 Pixels with events: {sample_events["h3_id"].nunique():,}')
    print(f'   📅 Date range: {sample_events["date"].min()} to {sample_events["date"].max()}')
    print(f'   🌩️ Extreme events: {len(sample_events[sample_events["magnitude"] == "extreme"]):,}')
    print(f'   ⛈️ Severe events: {len(sample_events[sample_events["magnitude"] == "severe"]):,}')
    print(f'   ⚡ Total time: {total_time/60:.1f} minutes')
    
    # Yearly breakdown
    yearly_events = sample_events.groupby('year').size()
    print(f'\n📅 Events by year (top 10):')
    for year, count in yearly_events.head(10).items():
        print(f'   {year}: {count:,} events')

# Clean up
del all_pixel_data

# Cell 2: Identify Extreme Rainfall Events from CHIRPS
"""
Process CHIRPS daily rainfall data to identify extreme events.
Uses objective thresholds and merges consecutive wet days into events.
"""

import rasterio
import numpy as np
from datetime import datetime, timedelta
import pandas as pd
from pathlib import Path

print('🌧️ EXTREME RAINFALL EVENT IDENTIFICATION')
print('=' * 45)

# Configuration
EVENTS_DIR = PREPROCESSED_DIR / 'events'
CHIRPS_FILES = sorted(CLIMATE_DIR.glob('zimbabwe_chirps_daily_*.tif'))

print(f'📊 Processing {len(CHIRPS_FILES)} CHIRPS files...')
print(f'🎯 Thresholds: {EXTREME_PERCENTILES}th percentiles + {MIN_EVENT_RAINFALL}mm/day')

# Check if events already processed
events_file = EVENTS_DIR / 'extreme_events_catalog.parquet'
if events_file.exists():
    print('✅ Extreme events catalog already exists')
    events_df = pd.read_parquet(events_file)
    print(f'   📋 Found {len(events_df)} extreme events')
    print(f'   📅 Period: {events_df.start_date.min()} to {events_df.end_date.max()}')
else:
    print('🔍 Identifying extreme rainfall events...')
    
    all_events = []
    processed_years = 0
    
    # Process each CHIRPS file (annual)
    for chirps_file in CHIRPS_FILES:
        try:
            # Extract year from filename
            year = int(chirps_file.stem.split('_')[-1])
            print(f'   Processing {year}...', end=' ')
            
            with rasterio.open(chirps_file) as src:
                # Read all bands (daily data for the year)
                rainfall_data = src.read()  # Shape: (days, height, width)
                transform = src.transform
                crs = src.crs
                
                # Get spatial dimensions
                n_days, height, width = rainfall_data.shape
                
                # Calculate percentile thresholds for this year
                # Flatten spatial data but keep temporal dimension
                flat_data = rainfall_data.reshape(n_days, -1)
                
                # Remove no-data values (typically 0 or negative)
                valid_mask = flat_data > 0
                
                # Calculate spatial percentiles for each day
                daily_percentiles = {}
                for pct in EXTREME_PERCENTILES:
                    daily_p = []
                    for day in range(n_days):
                        day_data = flat_data[day][valid_mask[day]]
                        if len(day_data) > 0:
                            daily_p.append(np.percentile(day_data, pct))
                        else:
                            daily_p.append(0)
                    daily_percentiles[pct] = np.array(daily_p)
                
                # Find extreme events for each pixel
                events_this_year = []
                
                # Sample processing on subset of pixels for speed
                sample_step = max(1, height * width // 10000)  # Sample ~10k pixels
                
                for i in range(0, height * width, sample_step):
                    row = i // width
                    col = i % width
                    
                    pixel_series = rainfall_data[:, row, col]
                    
                    # Skip if mostly no-data
                    if np.sum(pixel_series > 0) < n_days * 0.1:
                        continue
                    
                    # Identify extreme days
                    extreme_days = []
                    for day in range(n_days):
                        rainfall = pixel_series[day]
                        
                        # Check if meets any threshold
                        is_extreme = (rainfall >= MIN_EVENT_RAINFALL and 
                                    any(rainfall >= daily_percentiles[pct][day] 
                                        for pct in EXTREME_PERCENTILES))
                        
                        if is_extreme:
                            extreme_days.append({
                                'day': day,
                                'date': datetime(year, 1, 1) + timedelta(days=day),
                                'rainfall': rainfall,
                                'row': row,
                                'col': col
                            })
                    
                    # Group consecutive days into events
                    if extreme_days:
                        current_event = [extreme_days[0]]
                        
                        for extreme_day in extreme_days[1:]:
                            # Check if consecutive (within 2 days)
                            if extreme_day['day'] - current_event[-1]['day'] <= 2:
                                current_event.append(extreme_day)
                            else:
                                # Save current event
                                if len(current_event) > 0:
                                    events_this_year.append({
                                        'year': year,
                                        'start_date': current_event[0]['date'],
                                        'end_date': current_event[-1]['date'],
                                        'duration_days': len(current_event),
                                        'total_rainfall': sum(d['rainfall'] for d in current_event),
                                        'peak_intensity': max(d['rainfall'] for d in current_event),
                                        'center_row': current_event[0]['row'],
                                        'center_col': current_event[0]['col']
                                    })
                                
                                # Start new event
                                current_event = [extreme_day]
                        
                        # Don't forget last event
                        if len(current_event) > 0:
                            events_this_year.append({
                                'year': year,
                                'start_date': current_event[0]['date'],
                                'end_date': current_event[-1]['date'], 
                                'duration_days': len(current_event),
                                'total_rainfall': sum(d['rainfall'] for d in current_event),
                                'peak_intensity': max(d['rainfall'] for d in current_event),
                                'center_row': current_event[0]['row'],
                                'center_col': current_event[0]['col']
                            })
                
                all_events.extend(events_this_year)
                processed_years += 1
                print(f'{len(events_this_year)} events')
                
        except Exception as e:
            print(f'❌ Error processing {year}: {e}')
            continue
    
    # Convert to DataFrame and save
    if all_events:
        events_df = pd.DataFrame(all_events)
        events_df.to_parquet(events_file)
        
        print(f'\n✅ Event identification complete!')
        print(f'   📋 Total events: {len(events_df)}')
        print(f'   📅 Years processed: {processed_years}')
        print(f'   💾 Saved: {events_file}')
        
        # Event statistics
        print(f'\n📊 Event Statistics:')
        print(f'   Duration: {events_df.duration_days.mean():.1f} ± {events_df.duration_days.std():.1f} days')
        print(f'   Peak intensity: {events_df.peak_intensity.mean():.1f} ± {events_df.peak_intensity.std():.1f} mm/day')
        print(f'   Total rainfall: {events_df.total_rainfall.mean():.1f} ± {events_df.total_rainfall.std():.1f} mm')
    else:
        print('❌ No extreme events identified')

print(f'\n📁 Output: {events_file}')

# Cell 5: Condition Events by Climate Phases (ENSO/IOD)
"""
Link extreme rainfall events to climate indices for frequency modeling.
Determines which climate phase each event occurred during.
"""

import pandas as pd
import numpy as np
from datetime import datetime

print('🌍 CLIMATE PHASE CONDITIONING')
print('=' * 32)

# File paths
events_file = EVENTS_DIR / 'extreme_events_catalog.parquet'
climate_file = INDICES_DIR / 'zimbabwe_climate_indices.csv'
conditioned_events_file = PREPROCESSED_DIR / 'events' / 'climate_conditioned_events.parquet'

# Load data
print('📊 Loading events and climate indices...')
events_df = pd.read_parquet(events_file)
climate_df = pd.read_csv(climate_file)

print(f'   Events: {len(events_df):,}')
print(f'   Climate records: {len(climate_df):,}')

# Check if conditioning already done
if conditioned_events_file.exists():
    print('✅ Climate conditioning already completed')
    conditioned_df = pd.read_parquet(conditioned_events_file)
    print(f'   📋 Conditioned events: {len(conditioned_df):,}')
else:
    print('🔗 Linking events to climate phases...')
    
    # Prepare climate data for merging
    climate_df['date'] = pd.to_datetime(climate_df['date'])
    climate_df['year'] = climate_df['date'].dt.year
    climate_df['month'] = climate_df['date'].dt.month
    
    # Define climate phases
    def classify_enso_phase(nino34_sst):
        if pd.isna(nino34_sst) or nino34_sst == -999:
            return 'Unknown'
        elif nino34_sst >= 0.5:
            return 'El Niño'
        elif nino34_sst <= -0.5:
            return 'La Niña'
        else:
            return 'Neutral'
    
    def classify_iod_phase(dmi):
        if pd.isna(dmi) or dmi == -999:
            return 'Unknown'
        elif dmi >= 0.4:
            return 'Positive IOD'
        elif dmi <= -0.4:
            return 'Negative IOD'
        else:
            return 'Neutral'
    
    climate_df['enso_phase'] = climate_df['nino34_sst'].apply(classify_enso_phase)
    climate_df['iod_phase'] = climate_df['dmi'].apply(classify_iod_phase)
    
    # Add event timing info
    events_df['start_date'] = pd.to_datetime(events_df['start_date'])
    events_df['start_month'] = events_df['start_date'].dt.month
    
    # Merge events with climate data
    conditioned_events = []
    
    for _, event in events_df.iterrows():
        event_year = event['year']
        event_month = event['start_month']
        
        # Find matching climate record
        climate_match = climate_df[
            (climate_df['year'] == event_year) & 
            (climate_df['month'] == event_month)
        ]
        
        if len(climate_match) > 0:
            climate_record = climate_match.iloc[0]
            
            # Create conditioned event record
            conditioned_event = {
                **event.to_dict(),
                'nino34_sst': climate_record['nino34_sst'],
                'dmi': climate_record['dmi'],
                'enso_phase': climate_record['enso_phase'],
                'iod_phase': climate_record['iod_phase'],
                'climate_composite': f"{climate_record['enso_phase']}_{climate_record['iod_phase']}"
            }
            conditioned_events.append(conditioned_event)
    
    # Save conditioned events
    if conditioned_events:
        conditioned_df = pd.DataFrame(conditioned_events)
        conditioned_df.to_parquet(conditioned_events_file)
        
        print(f'✅ Climate conditioning complete!')
        print(f'   📋 Conditioned events: {len(conditioned_df):,}')
        print(f'   💾 Saved: {conditioned_events_file}')
        
        # Climate phase statistics
        print(f'\n📊 Events by Climate Phase:')
        print(f'   ENSO phases:')
        for phase, count in conditioned_df['enso_phase'].value_counts().items():
            print(f'     {phase}: {count:,} events ({count/len(conditioned_df)*100:.1f}%)')
        
        print(f'   IOD phases:')
        for phase, count in conditioned_df['iod_phase'].value_counts().items():
            print(f'     {phase}: {count:,} events ({count/len(conditioned_df)*100:.1f}%)')
        
        # Most active climate combinations
        print(f'\n🎯 Top climate combinations:')
        top_combos = conditioned_df['climate_composite'].value_counts().head()
        for combo, count in top_combos.items():
            print(f'   {combo}: {count:,} events')
    
    else:
        print('❌ No events could be conditioned')

print(f'\n📁 Output: {conditioned_events_file}')

def latest_required(folder: Path, pattern: str) -> Path:

def read_json(path: Path) -> dict:
    assert path.exists(), f"Missing JSON: {path}"
    return json.loads(path.read_text(encoding="utf-8"))


def write_json(path: Path, payload: dict):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload, indent=2), encoding="utf-8")


def sha256_file(path: Path) -> str:
    h = hashlib.sha256()

    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)

    return h.hexdigest()


def validate_checksums(bundle_dir: Path, checksums_payload: dict):
    rows = []
    ok = True

    for rec in checksums_payload.get("files", []):
        rel = rec["relative_path"]
        expected = rec["sha256"]
        path = bundle_dir / rel

        exists = path.exists()
        actual = sha256_file(path) if exists else None
        match = bool(exists and actual == expected)

        if not match:
            ok = False

        rows.append({
            "relative_path": rel,
            "exists": exists,
            "sha256_match": match,
            "expected_sha256": expected,
            "actual_sha256": actual,
        })

    return ok, rows


def json_records(df_part: pd.DataFrame):
